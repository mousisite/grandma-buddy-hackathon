# Grandma Buddy packages (see docs/ARCHITECTURE.md).
# All of these are free, open-source packages. Most have permissive licenses
# (MIT, Apache-2.0, or BSD); the voice's pronunciation helpers are GPL-3.0
# (marked below). They are installed by pip, never copied into this repo.
# Install them with: .venv\Scripts\python -m pip install -r requirements.txt
#
# Buddy's natural voice (optional; without it Buddy uses the browser's voice):
#   .venv\Scripts\python -m pip install kokoro-onnx==0.6.1 --no-deps --ignore-requires-python
#   .venv\Scripts\python scripts\get_voice.py
# (kokoro-onnx is MIT and plain Python; the flag is only because it isn't
# marked for Python 3.14 yet. Its other packages are listed below.)

fastapi            # our web server framework (MIT)
uvicorn[standard]  # runs the FastAPI app (BSD-3-Clause)
python-multipart   # lets the server receive uploaded files, like voice recordings (Apache-2.0)
openai             # official OpenAI package; we point it at Meta's API for chat (Apache-2.0)
requests           # sends voice files to Muse Voice Transcribe (Apache-2.0)
python-dotenv      # reads settings from the .env file (BSD-3-Clause)
pytest             # runs our tests (MIT)
httpx              # needed by FastAPI's test client (BSD-3-Clause)
playwright         # checks our screens in a real browser (Apache-2.0)
onnxruntime        # runs Buddy's natural voice model on this computer (MIT)
numpy              # the voice model's sound, turned into a WAV (BSD-3-Clause)
phonemizer         # how words are pronounced, for the voice (GPL-3.0)
espeakng-loader    # brings espeak-ng, the pronunciation dictionary phonemizer uses (GPL-3.0)
